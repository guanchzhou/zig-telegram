echo_bot.zig — Zig 0.17.0 
const std = @import("std");
const telegram = @import("telegram");

pub fn main(init: std.process.Init) !void {
    var http: telegram.transport.HttpTransport = .init(init.gpa, init.io);
    defer http.deinit();

    const token = try telegram.SecretToken.initValidated("123456:ABC-DEF");
    var client = telegram.bot.BotClient.init(
        init.gpa, init.io, token, http.transport(), .{},
    );

    var detail: ?telegram.ErrorDetail = null;
    defer if (detail) |*d| d.deinit(init.gpa);

    var msg = try client.sendMessage(.{
        .chat_id = .fromInt(42),
        .text = "Hello from zig-telegram (Bot API 10.3)!",
        .parse_mode = .MarkdownV2,
    }, &detail);
    defer msg.deinit();
}
01 // THREE-PLANE ARCHITECTURE

One Module. Three Protocol Planes. Zero Hidden State.

Every network call, allocator, and error diagnostic is explicit. Choose high-level Bot API 10.3 ergonomics, low-level MTProto 2.0 binary framing, or dynamic TDLib JSON bindings.

PLANE 01 // BOT API 10.3

Recursive attach:// Multipart

Existing Zig libraries only inspect top-level method fields for InputFile, silently breaking sendMediaGroup, sendPaidMedia, and editMessageMedia.

  • Walks structs, optionals, slices, and unions
  • Emits "attach://part_<n>" in JSON automatically
  • Streams binary parts with CSPRNG 128-bit boundaries
  • RFC 7578 filename quote/CRLF escaping
PLANE 02 // SECURITY & RELIABILITY

SecretToken & Caller-Owned Errors

Bot tokens never leak in logs or stack traces, and concurrent threads never race on a shared mutable last_error struct field.

  • SecretToken.format prints bot<id>:[REDACTED]
  • Constant-time SHA-256 webhook header verification
  • Explicit err_out: ?*?ErrorDetail with retry_after
  • Automatic --local Bot API server path remapping
PLANE 03 // MTPROTO 2.0

Pure-Zig Cryptography & Framing

Full MTProto 2.0 binary packet codec built exclusively on Zig's standard library primitives.

  • AES-256-IGE mode on top of std.crypto.core.aes.Aes256
  • MTProto 2.0 SHA-256 msg_key & AES key/IV derivation
  • Abridged, Intermediate, Padded & Obfuscated2 framing
  • 64-bit time-synchronized monotonic msg_id generator

const std = @import("std");
const telegram = @import("telegram");

const AppCtx = struct {
    client: *telegram.bot.BotClient,
    pub fn onStart(self: *@This(), msg: telegram.bot.types.Message, args: []const u8) !void {
        _ = args;
        var r = try self.client.sendMessage(.{
            .chat_id = .fromInt(msg.chat.id),
            .text = "Welcome to zig-telegram!",
        }, null);
        r.deinit();
    }
};

const AppRouter = telegram.bot.Router(*AppCtx, .{
    .commands = &.{.{ .name = "start", .handler = AppCtx.onStart }},
});
